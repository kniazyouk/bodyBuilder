          </div>

            {title}
          </h3>

          <ul className="flex flex-col gap-2 mt-auto">
            {items.map((item) => (
              <li key={item} className="text-muted text-sm flex items-center gap-2">
                <span className={isOrange ? "text-neon-orange" : "text-neon-cyan"}>
                  +
                </span>

      </div>

┌────────────────────────────────────────────────────────────┐
│ React #4: ServiceCard NO TIENE DATOS: los recibe por PROPS │

│ ¿De dónde saca los datos? -> de App.tsx. Los datos están  │
│ hardcodeados en App.tsx y se pasan como props a cada      │

│                                                           │
│ interface ServiceCardProps tipa las 4 props:              │
│   accent: "orange" | "cyan"  union type: solo admite esos│
│                               dos valores. Si pasás otro, │

│                               App.tsx evalúa <CoachIcon   │

══════════════════════════════════════════════════════════════════

  #1  Componente funcional               -> Navbar, Hero, etc.
  #2  .map() para listas                 -> links, testimonios, videos

  #4  Props tipadas + ternario           -> ServiceCard (accent)

  #6  iframes + template literal         -> VideoGallery (src dinámica)
  #7  Array de objetos en JSX            -> socials del footer
  #8  Composición de componentes         -> App.tsx como orquestador
  #9  useState (estado local)            -> link activo del navbar
